```
# Pool and per-connection bounds for the index client (index-store.ts). The
# defaults are what production should run; before them the client was
# `new SQL(url)` — Bun's pool of 10, no connect timeout, no statement_timeout,
# an empty application_name. Out-of-range values clamp; unparseable fall back.
# Pool size per process, 1..50.
SUBSTRATE_PG_POOL_MAX=8
# One connect may take this long, 200..60000 ms.
SUBSTRATE_PG_CONNECT_TIMEOUT_MS=2000
# statement_timeout and idle_in_transaction_session_timeout, applied as startup
# parameters on every connection. 0 leaves the server default (unbounded).
SUBSTRATE_PG_STATEMENT_TIMEOUT_MS=5000
# Idle connections are closed after this many seconds; 0 keeps them forever.
SUBSTRATE_PG_IDLE_TIMEOUT_S=300

# ── Index retention sweep (index-sweep.ts; substrate-index-sweep.timer) ──────
# dry-run reports what the documented retention would remove and deletes
# nothing; apply deletes; off exits without reading the index. The CLI's
# `substrate index sweep --apply` is refused unless this is also apply — two
# keys to delete. Set on the sweep unit's drop-in, not in the server's env.
SUBSTRATE_INDEX_SWEEP=dry-run
# Days before a row is a candidate, per class (1..3650). Defaults are the
# documented retention: terminal graphs + 90 d, superseded versions 90 d,
# spent derived idempotency keys 7 d, explicit keys 30 d (exactly-once holds
# that long), decided reviews 30 d, quarantine whose content later landed 30 d.
SUBSTRATE_SWEEP_GRAPHS_DAYS=90
SUBSTRATE_SWEEP_SUPERSEDED_DAYS=90
SUBSTRATE_SWEEP_DERIVED_WRITES_DAYS=7
SUBSTRATE_SWEEP_EXPLICIT_WRITES_DAYS=30
SUBSTRATE_SWEEP_REVIEWS_DAYS=30
SUBSTRATE_SWEEP_QUARANTINE_DAYS=30
# Rows per DELETE statement, 1..5000; each statement stays inside
# SUBSTRATE_PG_STATEMENT_TIMEOUT_MS and skips rows a live writer holds.
SUBSTRATE_SWEEP_BATCH=500
```
